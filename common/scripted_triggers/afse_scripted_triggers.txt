afse_can_research_arkship_technologies = {
	OR = {
		# Preserve the vanilla technology path for Nomadic empires.
		is_nomadic = yes

		# Settled empires require a dedicated hybrid Arkship doctrine.
		AND = {
			is_nomadic = no
			has_ascension_perk = ap_afse_arkship_continuity
		}
	}
}

afse_is_settled_arkship_empire = {
	is_nomadic = no
	has_ascension_perk = ap_afse_arkship_continuity
}
