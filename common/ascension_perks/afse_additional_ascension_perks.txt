# Additional vanilla Ascension Perks with Nomadic effects retained by Arkship-capable Settled empires.

ap_imperial_prerogative = {
	triggered_modifier = {
		potential = { is_nomadic = no }
		empire_size_colonies_mult = -0.25
		country_official_cap_add = 2
	}

	triggered_modifier = {
		potential = { is_nomadic = yes }
		pop_cat_ruler_bonus_workforce_mult = 0.15
		country_official_cap_add = 2
	}

	potential = {
		NOT = {
			has_ascension_perk = ap_imperial_prerogative
		}
		is_megacorp = no
	}

	ai_weight = {
		factor = 15
		modifier = {
			factor = 0
			num_owned_planets < 5
		}
		modifier = {
			factor = 0.1
			num_owned_planets < 7
		}
		modifier = {
			factor = 2
			num_owned_planets > 20
		}
		modifier = {
			factor = 5
			num_owned_planets > 40
		}
	}

	on_enabled = {
		run_ai_strategic_data = yes
	}

	triggered_modifier = {
		potential = { afse_is_settled_arkship_empire = yes }
		pop_cat_ruler_bonus_workforce_mult = 0.15
	}
}
