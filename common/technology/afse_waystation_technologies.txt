# Vanilla Waystation technologies with AFSE availability and effects for settled empires.

tech_waystation_1 = {
	cost = 0
	area = engineering
	start_tech = yes
	category = { voidcraft }
	prerequisites = { "tech_space_construction" }
	tier = 0

	prereqfor_desc = {
		ship = {
			title = "allow_waystations"
			desc = "waystation_megastructure_MEGASTRUCTURE_DETAILS"
		}
	}

	potential = {
		OR = {
			is_nomadic = yes
			afse_is_settled_arkship_empire = yes
		}
	}

	starting_potential = {
		is_nomadic = yes
	}

	modifier = {
		country_starbase_capacity_add = 2
	}

	technology_swap = {
		name = tech_waystation_1
		inherit_icon = yes
		inherit_effects = no
		trigger = {
			afse_is_settled_arkship_empire = yes
		}
		modifier = { }
	}

	ai_weight = {
		factor = @ai_starbase_types_factor
	}
}

tech_waystation_2 = {
	area = engineering
	category = { voidcraft }
	tier = 2
	cost = @tier2cost2
	weight = @tier2weight2
	prerequisites = { "tech_waystation_1" }

	potential = {
		OR = {
			is_nomadic = yes
			afse_is_settled_arkship_empire = yes
		}
	}

	prereqfor_desc = {
		hide_prereq_for_desc = ship
		custom = {
			title = "TECH_UNLOCK_WAYSTATION_2_CONSTRUCTION_TITLE"
			desc = "TECH_UNLOCK_WAYSTATION_2_CONSTRUCTION_DESC"
		}
	}

	modifier = {
		country_starbase_capacity_add = 4
		nomadic_constructor_limit_add = 2
		shipsize_nomads_constructor_stockpile_capacity_mult = 0.25
	}

	technology_swap = {
		name = tech_waystation_2
		inherit_icon = yes
		inherit_effects = no

		trigger = {
			is_nomadic = yes
			is_eager_explorer_empire = yes
		}
		modifier = {
			country_starbase_capacity_add = 4
			nomadic_constructor_limit_add = 2
			shipsize_nomads_constructor_stockpile_capacity_mult = 0.25
			shipsize_nomads_engineer_vessel_stockpile_capacity_mult = 0.25
		}
	}

	technology_swap = {
		name = tech_waystation_2
		inherit_icon = yes
		inherit_effects = no
		trigger = {
			afse_is_settled_arkship_empire = yes
		}
		modifier = {
			nomadic_constructor_limit_add = 2
			shipsize_nomads_constructor_stockpile_capacity_mult = 0.25
		}
	}

	weight_modifier = {
		modifier = {
			factor = 0.1
			NOT = { years_passed > 20 }
		}
		modifier = {
			factor = 10
			any_neighbor_country = {
				has_technology = tech_starbase_3
			}
		}
		modifier = {
			factor = 2
			years_passed > 30
		}
		modifier = {
			factor = 3
			years_passed > 35
		}
		modifier = {
			factor = 4
			years_passed > 40
		}
		modifier = {
			factor = 5
			count_starbase_sizes = {
				starbase_size = starbase_waystation_1
				count >= 3
			}
		}
		modifier = {
			factor = 1.25
			has_tradition = tr_expansion_adopt
		}
	}

	ai_weight = {
		factor = @ai_starbase_types_factor
	}
}
