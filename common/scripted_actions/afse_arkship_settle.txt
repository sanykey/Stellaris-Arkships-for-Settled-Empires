arkship_settle = { #Do not rename - name referenced in code
	user_scope = fleet
	scope = planet

	on_completed = afse_on_arkship_settle
	tooltip = arkship_settle_ship_button

	important = yes

	slot = 0

	hide_from_context_menu = {
		# Nomadic empires continue using the Expansion Planner.
		from.owner = {
			is_nomadic = yes
		}
	}
	possible = {
		if = {
			limit = {
				solar_system = {
					is_surveyed = {
						who = from
						status = no
					}
				}
			}
			custom_tooltip = {
				fail_text = arkship_target_not_surveyed_fail_tt
				always = no
			}
		}
		else = {
			custom_tooltip = {
				fail_text = arkship_target_has_owner_fail_tt
				NOT = {
					exists = owner_or_space_owner
				}
			}
			custom_tooltip = {
				fail_text = arkship_target_colonizable_world_fail_tt
				is_colonizable = yes
			}
			custom_tooltip = {
				fail_text = arkship_owner_is_subject_tt
				from.owner = { is_subject = no }
			}
		}
	}
	button_visible = {
		always = no
	}
	button_clickable = {
		owner = {
			nomad_can_settle_trigger = yes
		}
	}

	activity_key = arkship_settle_activity_key
	context_menu_name = arkship_settle
	required_progress = value:arkship_settle_required_progress_value
	progress_activity_key = arkship_settle_progress_activity_key
	required_progress_key = arkship_settle_required_progress_key
	cooldown = @arkship_settle_cooldown
	category = arkship_settle
	category_limit = 1
	cost = {
		influence = @arkship_settle_cost
	}
}
